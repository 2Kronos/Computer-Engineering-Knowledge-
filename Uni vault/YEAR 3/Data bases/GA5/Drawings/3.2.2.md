
# Create code  for graph
```CPHER

// Create Infrastructure Nodes
CREATE (pump1:WaterPump {
    id: "WP-001",
    name: "Main Station Pump",
    status: "active",
    capacity_gpm: 1200,
    install_date: date('2020-05-15')
})

CREATE (pump2:WaterPump {
    id: "WP-002", 
    name: "Secondary Pump",
    status: "active",
    capacity_gpm: 800,
    install_date: date('2021-03-20')
})

CREATE (intersection:StreetIntersection {
    id: "SI-045",
    name: "Main & 5th Intersection",
    traffic_volume: "high",
    coordinates: point({x: -73.935242, y: 40.730610})
})

CREATE (request:MaintenanceRequest {
    id: "MR-2024-001",
    description: "Pressure fluctuation reported",
    priority: "high",
    created_date: datetime(),
    status: "open"
})

// Create Relationships between Nodes
CREATE (pump1)-[connects:CONNECTS_TO {
    pipe_diameter: "12 inches",
    material: "PVC",
    install_date: date('2020-05-15'),
    max_pressure: 150
}]->(pump2)

CREATE (pump1)-[requires:REQUIRES_MAINTENANCE {
    priority_level: 1,
    estimated_hours: 4,
    assigned_crew: "Crew-B",
    scheduled_date: date('2024-01-25')
}]->(request)

CREATE (pump2)-[located_at:LOCATED_AT {
    distance_meters: 15.5,
    access_type: "direct"
}]->(intersection)
```

# Display graph

```cypher
// Query to return all nodes and relationships for visualization
MATCH (n)-[r]->(m)
RETURN n, r, m
LIMIT 25
```
