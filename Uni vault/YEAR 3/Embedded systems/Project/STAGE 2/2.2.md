# Code 

```cpp
const int VALVE_PIN = 8;         // Blue LED - Water Inlet Valve
const int PUMP_PIN = 9;          // DC Motor - Water Pump
const int WATER_SENSOR_PIN = A5; // Potentiometer - Water Level Sensor
const int START_BUTTON_PIN = 10; // Push Button - Start/Pause/Resume

int currentWaterLevel = 0;
bool washingActive = false;
bool washingPaused = false;
int sequenceStep = 0;            // 0=waiting, 1=filling, 2=draining, 3=complete
unsigned long previousMillis = 0;
const long interval = 1000;      // Update every 1 second
bool displayNeeded = true;       // Flag to control display
bool buttonPressed = false;

void setup() {
  pinMode(VALVE_PIN, OUTPUT);
  pinMode(PUMP_PIN, OUTPUT);
  pinMode(WATER_SENSOR_PIN, INPUT);
  pinMode(START_BUTTON_PIN, INPUT_PULLUP);
  
  digitalWrite(VALVE_PIN, LOW);
  digitalWrite(PUMP_PIN, LOW);
  
  Serial.begin(9600);
  Serial.println("AUTOMATIC WASHING MACHINE SIMULATION");
  Serial.println("Press START button to begin");
  Serial.println("Press again to PAUSE, again to RESUME");
  Serial.println("---------------------------------------------");
}

void loop() {
  unsigned long currentMillis = millis();
  
  // Check for button press with debouncing
  if (digitalRead(START_BUTTON_PIN) == LOW) {
    if (!buttonPressed) {
      buttonPressed = true;
      handleButtonPress();
      delay(300); // Debounce delay
    }
  } else {
    buttonPressed = false;
  }
  
  if (washingActive && !washingPaused) {
    // Update simulated water level every second
    if (currentMillis - previousMillis >= interval) {
      previousMillis = currentMillis;
      updateWaterLevel(); // Automatically changes water level
      displayNeeded = true; // Display after each update
    }
    runWashSequence();
  }
  
  // Only display when needed (after changes or button press)
  if (displayNeeded) {
    displayStatus();
    displayNeeded = false; // Reset flag until next change
  }
  
  delay(100); // Small delay to avoid flooding
}

// Handle button press (Start/Pause/Resume)
void handleButtonPress() {
  if (!washingActive || sequenceStep == 3) { // IDLE or COMPLETE state
    // Start NEW wash cycle from beginning
    startWashSequence();
    Serial.println("START BUTTON: New Cycle STARTED");
  } 
  else if (washingActive && !washingPaused) {
    // Pause current cycle
    washingPaused = true;
    digitalWrite(VALVE_PIN, LOW);
    digitalWrite(PUMP_PIN, LOW);
    Serial.println("START BUTTON: Cycle PAUSED");
  }
  else if (washingPaused) {
    // Resume paused cycle
    washingPaused = false;
    Serial.println("START BUTTON: Cycle RESUMED");
  }
  displayNeeded = true;
}

// SIMULATED water level changes
void updateWaterLevel() {
  if (sequenceStep == 1 && !washingPaused) { // Filling
    currentWaterLevel += 5; // Increase water level
    if (currentWaterLevel > 100) currentWaterLevel = 100;
  } else if (sequenceStep == 2 && !washingPaused) { // Draining
    currentWaterLevel -= 5; // Decrease water level
    if (currentWaterLevel < 0) currentWaterLevel = 0;
  }
}

void startWashSequence() {
  washingActive = true;
  washingPaused = false;
  sequenceStep = 1; // Start with filling
  currentWaterLevel = 0; // Start with empty drum
}

void runWashSequence() {
  if (washingPaused) return; // Don't run if paused
  
  switch(sequenceStep) {
    case 1: // FILLING
      if (fillDrum()) {
        sequenceStep = 2; // Move to draining
        displayNeeded = true;
      }
      break;
      
    case 2: // DRAINING
      if (draining()) {
        sequenceStep = 3; // Complete
        washingActive = false;
        washingPaused = false;
        displayNeeded = true;
        Serial.println("WASH CYCLE COMPLETE!");
      }
      break;
  }
}

// REQUIRED FUNCTION: fillDrum() - Fill to 80%
bool fillDrum() {
  if (!washingPaused) {
    digitalWrite(VALVE_PIN, HIGH); // Valve OPEN
  }
  
  if (currentWaterLevel >= 80) {
    digitalWrite(VALVE_PIN, LOW); // Valve CLOSED
    Serial.println("DRUM FULL (80%) - Valve closed");
    Serial.println("Starting drain...");
    return true;
  }
  return false;
}

// REQUIRED FUNCTION: draining() - Drain to 10%
bool draining() {
  if (!washingPaused) {
    digitalWrite(PUMP_PIN, HIGH); // Pump ON
  }
  
  if (currentWaterLevel <= 10) {
    digitalWrite(PUMP_PIN, LOW); // Pump OFF
    Serial.println("DRUM EMPTY (10%) - Pump stopped");
    return true;
  }
  return false;
}

void displayStatus() {
  Serial.print("Water Level: ");
  Serial.print(currentWaterLevel);
  Serial.println("%");
  
  Serial.print("Valve: ");
  Serial.println(digitalRead(VALVE_PIN) ? "OPEN" : "CLOSED");
  
  Serial.print("Pump: ");
  Serial.println(digitalRead(PUMP_PIN) ? "ON" : "OFF");
  
  if (washingActive) {
    if (washingPaused) {
      Serial.println("Status: PAUSED - Press to RESUME");
    } else {
      Serial.print("Status: ");
      Serial.println(sequenceStep == 1 ? "FILLING" : "DRAINING");
    }
  } else {
    if (sequenceStep == 3) {
      Serial.println("Status: COMPLETE - Press to START NEW CYCLE");
    } else {
      Serial.println("Status: IDLE - Press to START");
    }
  }
  
  Serial.println("---------------------------------------------");
}
```

## **Core Functions: fillDrum() & draining()**
```cpp
bool fillDrum() {
  if (!washingPaused) {
    digitalWrite(VALVE_PIN, HIGH); // Open valve - water flows in
  }
  
  if (currentWaterLevel >= 80) {
    digitalWrite(VALVE_PIN, LOW);  // Close valve - stop water
    Serial.println("DRUM FULL (80%) - Valve closed");
    Serial.println("Starting drain...");
    return true; // Filling complete
  }
  return false; // Still filling
}

bool draining() {
  if (!washingPaused) {
    digitalWrite(PUMP_PIN, HIGH); // Start pump - water drains out
  }
  
  if (currentWaterLevel <= 10) {
    digitalWrite(PUMP_PIN, LOW);  // Stop pump - draining complete
    Serial.println("DRUM EMPTY (10%) - Pump stopped");
    return true; // Draining complete
  }
  return false; // Still draining
}
```

## **Key Design Features:**
1. **State Machine Architecture** - Clear sequence steps
2. **Non-blocking delays** - Uses `millis()` instead of `delay()`
3. **Efficient display** - Only updates when needed
4. **Proper debouncing** - Prevents multiple button triggers
5. **Modular structure** - Easy to maintain and expand
